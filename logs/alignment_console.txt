#@title Verify both predictions and submit jFATCAT-rigid to RCSB
from pathlib import Path
import json, hashlib, math, requests
rf_path=Path('/content/row29_rosettafold2.pdb')
af_matches=list(Path('/content/af2-run/results').glob('*unrelaxed_rank_001*.pdb'))
assert len(af_matches)==1, af_matches
af_path=af_matches[0]
for path in [rf_path,af_path]:
    atoms=[l for l in path.read_text().splitlines() if l.startswith('ATOM')]
    ca=[l for l in atoms if l[12:16].strip()=='CA']
    assert len(ca)==100 and {l[21] for l in ca}=={'A'}
    assert all(math.isfinite(float(l[a:b])) for l in atoms for a,b in [(30,38),(38,46),(46,54)])
    print(path.name,'residues',len(ca),'atoms',len(atoms),'sha256',hashlib.sha256(path.read_bytes()).hexdigest(),flush=True)
query={'context':{'mode':'pairwise','method':{'name':'fatcat-rigid'},'structures':[{'format':'pdb','selection':{'asym_id':'A'}},{'format':'pdb','selection':{'asym_id':'A'}}]}}
with rf_path.open('rb') as f1, af_path.open('rb') as f2:
    response=requests.post('https://alignment.rcsb.org/api/v1/structures/submit',params={'query':json.dumps(query)},files=[('files',('rf2.pdb',f1,'chemical/x-pdb')),('files',('af2.pdb',f2,'chemical/x-pdb'))],timeout=60)
print('RCSB HTTP',response.status_code,response.text,flush=True)
response.raise_for_status()
ticket=response.text.strip().strip(chr(34))
Path('/content/row29_alignment_ticket.txt').write_text(ticket)
Path('/content/row29_alignment_request.json').write_text(json.dumps(query,indent=2))
print('Result page: https://www.rcsb.org/alignment?uuid='+ticket)
